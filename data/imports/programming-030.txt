def EISITIRIO(enilikes, paidia):
    kostos = (enilikes * 10) +( paidia * 5)
    return kostos

    paidia = input("Δώσε αριθμό παιδιών: ")
    if (enilikes + paidia) > eleftheres_theseis:

        synolo_enilikwn_theatwn = synolo_enilikwn_theatwn + enilikes
        synolo_paidiwn_theatwn = synolo_paidiwn_theatwn + paidia
