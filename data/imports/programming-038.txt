def ypologismos(tem):
    if tem>=1 and tem<=3:
        xr=tem*120
    elif tem>=4 and tem<=6:
        xr=3*120+(tem-3)*100
    else:
        xr=3*120+3*100+(tem-6)*70
    return xr
